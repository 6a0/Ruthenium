package net.ruthenium.mixin.render.entity;

import com.llamalad7.mixinextras.injector.wrapoperation.Operation;
import com.llamalad7.mixinextras.injector.wrapoperation.WrapOperation;
import com.mojang.blaze3d.vertex.PoseStack;
import com.mojang.blaze3d.vertex.VertexConsumer;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.Font;
import net.minecraft.client.gui.font.TextRenderable;
import net.minecraft.client.renderer.SubmitNodeCollector;
import net.minecraft.client.renderer.entity.DisplayRenderer;
import net.minecraft.client.renderer.entity.state.TextDisplayEntityRenderState;
import net.minecraft.util.FormattedCharSequence;
import net.ruthenium.Initializer;
import net.ruthenium.render.entity.TextDisplayCulling;
import org.joml.Matrix4f;
import org.joml.Vector3f;
import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Unique;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

/** TEMPORARY (-Druthenium.verifyTextDisplayCulling=true): every drawn vertex of a text display lies within the culling radius */
@Mixin(DisplayRenderer.TextDisplayRenderer.class)
public class TextDisplayCullingCheckM {
    @Unique private static final boolean ENABLED = Boolean.getBoolean("ruthenium.verifyTextDisplayCulling");
    @Unique private static float cx, cy, cz;
    @Unique private static double radius;
    @Unique private static long vertices, outside, displays;
    @Unique private static double maxRatio;
    @Unique private static long lastLog;

    @Inject(method = "submitInner(Lnet/minecraft/client/renderer/entity/state/TextDisplayEntityRenderState;Lcom/mojang/blaze3d/vertex/PoseStack;Lnet/minecraft/client/renderer/SubmitNodeCollector;IF)V",
            at = @At("HEAD"))
    private void ruthenium$begin(TextDisplayEntityRenderState state, PoseStack poseStack, SubmitNodeCollector collector, int light, float progress,
                                 CallbackInfo ci) {
        if (!ENABLED) {
            return;
        }
        // The pose here is translate(entity) * billboard * transformation; its origin is the transformation's translation,
        // so take the entity position from the state against the camera
        var camera = Minecraft.getInstance().gameRenderer.mainCamera().position();
        cx = (float) (state.x - camera.x);
        cy = (float) (state.y - camera.y);
        cz = (float) (state.z - camera.z);
        radius = TextDisplayCulling.radius(state.cachedInfo, state.renderState.transformation().get(progress));
        displays++;
    }

    @WrapOperation(method = "submitInner(Lnet/minecraft/client/renderer/entity/state/TextDisplayEntityRenderState;Lcom/mojang/blaze3d/vertex/PoseStack;Lnet/minecraft/client/renderer/SubmitNodeCollector;IF)V",
            at = @At(value = "INVOKE", target = "Lnet/minecraft/client/renderer/SubmitNodeCollector;submitText(Lcom/mojang/blaze3d/vertex/PoseStack;FFLnet/minecraft/util/FormattedCharSequence;ZLnet/minecraft/client/gui/Font$DisplayMode;IIII)V"))
    private void ruthenium$checkText(SubmitNodeCollector collector, PoseStack poseStack, float x, float y, FormattedCharSequence line, boolean shadow,
                                     Font.DisplayMode mode, int light, int color, int background, int outline, Operation<Void> original) {
        if (ENABLED) {
            Matrix4f pose = new Matrix4f(poseStack.last().pose());
            Font font = Minecraft.getInstance().font;
            VertexConsumer recorder = new Recorder();
            font.prepareText(line, x, y, color, shadow, false, background).visit(new Font.GlyphVisitor() {
                @Override
                public void acceptRenderable(TextRenderable renderable) {
                    renderable.render(pose, recorder, light, false);
                }
            });
            if (outline != 0) {
                font.prepare8xTextOutline(line, x, y, outline).visit(new Font.GlyphVisitor() {
                    @Override
                    public void acceptRenderable(TextRenderable renderable) {
                        renderable.render(pose, recorder, light, false);
                    }
                });
            }
            log();
        }
        original.call(collector, poseStack, x, y, line, shadow, mode, light, color, background, outline);
    }

    @WrapOperation(method = "submitInner(Lnet/minecraft/client/renderer/entity/state/TextDisplayEntityRenderState;Lcom/mojang/blaze3d/vertex/PoseStack;Lnet/minecraft/client/renderer/SubmitNodeCollector;IF)V",
            at = @At(value = "INVOKE", target = "Lnet/minecraft/client/renderer/SubmitNodeCollector;submitTextBackground(Lcom/mojang/blaze3d/vertex/PoseStack;FFFFILnet/minecraft/client/gui/Font$DisplayMode;I)V"))
    private void ruthenium$checkBackground(SubmitNodeCollector collector, PoseStack poseStack, float x0, float y0, float x1, float y1, int color,
                                           Font.DisplayMode mode, int light, Operation<Void> original) {
        if (ENABLED) {
            Matrix4f pose = poseStack.last().pose();
            Vector3f v = new Vector3f();
            check(pose.transformPosition(x0, y0, 0.0F, v));
            check(pose.transformPosition(x0, y1, 0.0F, v));
            check(pose.transformPosition(x1, y0, 0.0F, v));
            check(pose.transformPosition(x1, y1, 0.0F, v));
        }
        original.call(collector, poseStack, x0, y0, x1, y1, color, mode, light);
    }

    @Unique
    private static void check(Vector3f v) {
        double dx = v.x - cx, dy = v.y - cy, dz = v.z - cz;
        double ratio = Math.sqrt(dx * dx + dy * dy + dz * dz) / radius;
        vertices++;
        maxRatio = Math.max(maxRatio, ratio);
        if (ratio > 1.0) {
            outside++;
        }
    }

    @Unique
    private static void log() {
        long now = System.currentTimeMillis();
        if (now - lastLog > 5000) {
            lastLog = now;
            Initializer.LOGGER.info("Text display culling check: {} displays, {} vertices, {} outside the radius, max distance / radius {}",
                    displays, vertices, outside, maxRatio);
        }
    }

    @Unique
    private static final class Recorder implements VertexConsumer {
        private final Vector3f v = new Vector3f();

        @Override
        public VertexConsumer addVertex(float x, float y, float z) {
            check(this.v.set(x, y, z));
            return this;
        }

        @Override
        public VertexConsumer setColor(int r, int g, int b, int a) {
            return this;
        }

        @Override
        public VertexConsumer setColor(int color) {
            return this;
        }

        @Override
        public VertexConsumer setUv(float u, float v) {
            return this;
        }

        @Override
        public VertexConsumer setUv1(int u, int v) {
            return this;
        }

        @Override
        public VertexConsumer setUv2(int u, int v) {
            return this;
        }

        @Override
        public VertexConsumer setUv3(float u, float v) {
            return this;
        }

        @Override
        public VertexConsumer setNormal(float x, float y, float z) {
            return this;
        }

        @Override
        public VertexConsumer setLineWidth(float width) {
            return this;
        }
    }
}
