package net.ruthenium.render.text;

import com.mojang.blaze3d.vertex.VertexConsumer;
import net.minecraft.client.gui.Font;
import net.minecraft.client.gui.font.TextRenderable;
import net.minecraft.locale.Language;
import net.minecraft.network.chat.Component;
import net.minecraft.network.chat.FormattedText;
import net.minecraft.util.FormattedCharSequence;
import net.ruthenium.Initializer;
import org.joml.Matrix4f;

import java.util.ArrayList;
import java.util.List;

/** TEMPORARY: compares kept text layouts with fresh ones (-Druthenium.verifyTextCache=true) */
public final class TextCacheCheck {
    public static final boolean ENABLED = Boolean.getBoolean("ruthenium.verifyTextCache");
    private static long widths, widthMismatches, orders, orderMismatches, prepared, preparedMismatches;
    private static long lastLog;

    private TextCacheCheck() {
    }

    static void checkWidth(Font font, FormattedText text, int kept) {
        widths++;
        if (font.width(text) != kept) {
            widthMismatches++;
        }
        log();
    }

    /** A fresh name tag replaced by the kept equal one: both must measure and order the same */
    static void checkKeep(Component kept, Component fresh) {
        Font font = net.minecraft.client.Minecraft.getInstance().font;
        widths++;
        if (font.width(kept) != font.width(fresh)) {
            widthMismatches++;
        }
        orders++;
        if (!record(Language.getInstance().getVisualOrder(fresh)).equals(record(kept.getVisualOrderText()))) {
            orderMismatches++;
        }
        log();
    }

    static void checkPrepared(Font font, FormattedCharSequence line, float x, float y, int color, boolean shadow, boolean includeEmpty, int bg,
                              Font.PreparedText kept) {
        prepared++;
        Font.PreparedText fresh = font.prepareText(SignTextCache.unwrap(line), x, y, color, shadow, includeEmpty, bg);
        if (!render(fresh).equals(render(kept))) {
            preparedMismatches++;
        }
        log();
    }

    private static List<Object> record(FormattedCharSequence sequence) {
        List<Object> out = new ArrayList<>();
        sequence.accept((position, style, codepoint) -> {
            out.add(position);
            out.add(style);
            out.add(codepoint);
            return true;
        });
        return out;
    }

    private static List<Object> render(Font.PreparedText text) {
        List<Object> out = new ArrayList<>();
        Matrix4f pose = new Matrix4f().translate(1.5F, -2.25F, 3.0F).rotateY(0.7F).scale(0.025F, -0.025F, 0.025F);
        Recorder recorder = new Recorder(out);
        text.visit(new Font.GlyphVisitor() {
            @Override
            public void acceptRenderable(TextRenderable renderable) {
                out.add(renderable.renderType(Font.DisplayMode.NORMAL));
                out.add(renderable.renderType(Font.DisplayMode.SEE_THROUGH));
                renderable.render(pose, recorder, 0xF000F0, false);
            }
        });
        return out;
    }

    private static void log() {
        long now = System.currentTimeMillis();
        if (now - lastLog > 5000) {
            lastLog = now;
            Initializer.LOGGER.info("Text cache check: widths {} ({} mismatches), visual orders {} ({}), prepared {} ({})",
                    widths, widthMismatches, orders, orderMismatches, prepared, preparedMismatches);
        }
    }

    private record Recorder(List<Object> out) implements VertexConsumer {
        @Override
        public VertexConsumer addVertex(float x, float y, float z) {
            this.out.add(Float.floatToRawIntBits(x));
            this.out.add(Float.floatToRawIntBits(y));
            this.out.add(Float.floatToRawIntBits(z));
            return this;
        }

        @Override
        public VertexConsumer setColor(int r, int g, int b, int a) {
            this.out.add(r << 24 | g << 16 | b << 8 | a);
            return this;
        }

        @Override
        public VertexConsumer setColor(int color) {
            this.out.add(color);
            return this;
        }

        @Override
        public VertexConsumer setUv(float u, float v) {
            this.out.add(Float.floatToRawIntBits(u));
            this.out.add(Float.floatToRawIntBits(v));
            return this;
        }

        @Override
        public VertexConsumer setUv1(int u, int v) {
            this.out.add(u << 16 | v);
            return this;
        }

        @Override
        public VertexConsumer setUv2(int u, int v) {
            this.out.add(u << 16 | v);
            return this;
        }

        @Override
        public VertexConsumer setUv3(float u, float v) {
            return this.setUv(u, v);
        }

        @Override
        public VertexConsumer setNormal(float x, float y, float z) {
            return this.addVertex(x, y, z);
        }

        @Override
        public VertexConsumer setLineWidth(float width) {
            this.out.add(Float.floatToRawIntBits(width));
            return this;
        }
    }
}
